=== _ar_from_preflight in full (the read it performs) ===
_ar_from_preflight() { # <raw-uuid>
    local raw="$1" aid root path
    aid="$(_ar_canonical_uuid "$raw")"
    if [[ -z "$aid" ]]; then
        echo "Error: --from takes an archive UUID; '${raw}' is not one." >&2
        return 1
    fi
    if ! root="$(_ar_require_real_root)"; then
        echo "Error: no archive root at $(_ar_root)." >&2
        return 1
    fi
    path="${root}/${aid}"
    # A live claim means this exact id is being published or purged right now. Inheriting
    # from an archive mid-flight would freeze counts that are about to change or vanish.
    if [[ -e "${root}/.publishing.${aid}" ]]; then
        echo "Error: archive ${aid} is being published or purged right now (${root}/.publishing.${aid})." >&2
        return 1
    fi
    # A direct child of the archive root, and a real directory — never a link, which is
    # how a lineage pointer would otherwise reach outside the archive.
    if [[ ! -d "$path" || -L "$path" ]]; then
        echo "Error: no archive ${aid} in ${root}." >&2
        return 1
    fi
    _path_is_direct_child "$root" "$path" || {
        echo "Error: ${path} is not a direct child of ${root}." >&2
        return 1
    }
    local f
    for f in meta digest.md; do
        if [[ -L "${path}/${f}" || ! -f "${path}/${f}" || ! -r "${path}/${f}" ]]; then
            echo "Error: archive ${aid} has no readable ${f}." >&2
            return 1
        fi
    done
    _ar_validate_tree "$path" "$aid" || {
        echo "Error: archive ${aid} did not validate — refusing to inherit from it." >&2
        return 1
    }
    local h p
    h="$(_ar_meta_get "${path}/meta" handover_count)"
    p="$(_ar_meta_get "${path}/meta" pending_request_count)"
    if [[ ! "$h" =~ ^[0-9]+$ || ! "$p" =~ ^[0-9]+$ ]]; then
        echo "Error: archive ${aid} has incomplete counts (handover='${h}', pending='${p}')." >&2
        return 1
    fi
    # ONE OBSERVATION, FROZEN. The counts leave with the id they were proved against —
    # the caller re-reading them afterwards is a second observation of a file another
    # process may be deleting, and the child would then record counts that were never
    # proved (or empty ones defaulted to zero).
    printf '%s\t%s\t%s' "$aid" "$h" "$p"
    return 0
}
